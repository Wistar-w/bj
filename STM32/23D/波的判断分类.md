接下来要让机器能看图识别
>主要是那个只处理了单一峰值
我们需要建立一个“排行榜”

**也就是前N高峰**
那么我们就要准备位置
<span style="color:cyan">=========================================

**CW**只有一个极高的主峰，只要一个位置就可以了

**AM**需要 3 个位置（中间 1 根载波峰，左右各 1 根边带峰）

**FM**比较麻烦，根据调频指数（mf）的不同，频谱上会出现一连串等间距的峰（载波 + 无数边带）。为了拟合它的形状，通常需要 7~12 个位置。

**ASK**基础也是 3 个位置（如果发送 1010 方波序列，主能量集中在载波和左右第一边带）。

**FSK**需要 2 个位置（代表 0 和 1 的两个频率主峰）。如果带宽宽，还需要额外位置记录边带，通常准备 4 个位置 绰绰有余。

**PSK**载波被抑制，它不表现为“尖峰”，而是一大片隆起的“馒头形”包络。为了测它的带宽，需要记住包络最高点和两侧的几个点，差不多要 3~5 个位置。

**<span style="color:blue">那么我们方便的话,就要准备12个位置**
```C
#define MAX_PEAKS 12

int peak_idx[MAX_PEAKS] = {0};   // 存下标
float peak_mag[MAX_PEAKS] = {0}; // 存幅度
uint8_t peak_num = 0;            // 记录当前实际抓到了几个峰
```
改掉找峰就好了，但要处理小毛刺的问题，就是剩下的位置被底噪填了
```C
    // 1. 如果连一个峰都没找到，直接退出
    if (peak_num == 0) return;

    // 2. 设定动态门限（比如最大峰值的 15%）
    float threshold = peak_mag[0] * 0.15f;

            valid_peaks++;
        } else {
            break; // 因为数组是从大到小排的，一旦遇到小于门限的，后面的全不用看了
        }
```

### FM
```C
bool is_FM(void) {
    if (valid_peaks < 4) return false;

    // 1. 准备一个临时结构体数组，用于按频率排序
    // （这里假设你工程里还没用结构体，我们用两个数组来模拟）
    int   sorted_idx[12];
    float sorted_mag[12];
    
    // 拷贝前 valid_peaks 个数据
    for (int i = 0; i < valid_peaks; i++) {
        sorted_idx[i] = peak_idx[i];
        sorted_mag[i] = peak_mag[i];
    }

    // 2. 冒泡排序：按频率（下标）从小到大排
    // 注意：一定要把 mag 也带着一起交换！
    for (int i = 0; i < valid_peaks - 1; i++) {
        for (int j = 0; j < valid_peaks - i - 1; j++) {
            if (sorted_idx[j] > sorted_idx[j+1]) {
                // 交换 idx
                int temp_idx = sorted_idx[j];
                sorted_idx[j] = sorted_idx[j+1];
                sorted_idx[j+1] = temp_idx;
                
                // 同步交换 mag，防止数据错位！
                float temp_mag = sorted_mag[j];
                sorted_mag[j] = sorted_mag[j+1];
                sorted_mag[j+1] = temp_mag;
            }
        }
    }

    // 3. 等间距校验
    int spacing = sorted_idx[1] - sorted_idx[0];
    for (int i = 2; i < valid_peaks; i++) {
        int current_spacing = sorted_idx[i] - sorted_idx[i-1];
        // 允许 2 个 bin 的误差
        if (abs(current_spacing - spacing) > 2) {
            return false;
        }
    }

    // 4. 进阶校验：FM 的边带幅度通常是随距离衰减的
    // （载波可能最大，也可能因为贝塞尔函数零点变成最小）
    // 简单的排除法：如果所有峰幅度都一模一样，那大概率是方波/矩形脉冲，不是FM
    bool all_same_mag = true;
    for (int i = 1; i < valid_peaks; i++) {
        if (abs(sorted_mag[i] - sorted_mag[0]) > (sorted_mag[0] * 0.05f)) {
            all_same_mag = false;
            break;
        }
    }
    if (all_same_mag) return false; 

    return true; // 确认是 FM 梳状谱
}
```
### AM/2ASK
```C
bool is_AM(void) {
    if (valid_peaks == 3) {
        int carrier_idx = peak_idx[0];
        int left_idx = -1, right_idx = -1;
        float left_mag = 0, right_mag = 0;

        // 在排行榜里找载波左边和右边的次高峰
        for (int i = 1; i < 3; i++) {
            if (peak_idx[i] < carrier_idx) {
                left_idx = peak_idx[i];
                left_mag = peak_mag[i];
            } else {
                right_idx = peak_idx[i];
                right_mag = peak_mag[i];
            }
        }

        // 如果能成功找到左右边带
        if (left_idx != -1 && right_idx != -1) {
            int dist_left  = carrier_idx - left_idx;
            int dist_right = right_idx - carrier_idx;
            
            // 对称性检测
            if (abs(dist_left - dist_right) <= 2) {
                // 计算调幅系数
                AM_ma = (left_mag + right_mag) / peak_mag[0];
                return true;
            }
        }
    }
    return false;
}
```

### 2PSK
```C
bool is_PSK(void) {
    // PSK 往往特征不明显，通常用排除法，或者看主峰的带宽
    // 简单版：如果有 2 个峰，但它们离得非常近（距离 < 5 个bin）
    // 而 FSK 的两个峰通常离得比较远，这里可以做个区分
    if (valid_peaks == 2) {
        int distance = abs(peak_idx[0] - peak_idx[1]);
        if (distance < 5) {
            // 靠得极近的两个隆起，大概率是抑制载波的 PSK
            PSK_fre_max = (peak_idx[0] + peak_idx[1]) / 2.0f * FFT_deta;
            return true;
        }
    }
    return false;
}
```

```C
void Analyze_Signal(void) {
    /* 这里省略了处理小毛刺代码 */
    if (valid_peaks == 0) return;//确认有无信号

    //我觉得可以用switch
    switch (valid_peaks) {

            if (is_CW()) {

            }
            break;
            
        case 2:
            // 2个峰可能是 FSK，也可能是 PSK
            if (is_FSK()) {
                printf("检测到: FSK, 频偏: %.2f Hz\n", FSK_fre_jump);
            } else if (is_PSK()) {
                printf("检测到: PSK, 载频约: %.2f Hz\n", PSK_fre_max);
            }
            break;
            
        case 3:
            if (is_AM()) {
                printf("检测到: AM, 调制度: %.2f\n", AM_ma);
            }
            break;
            
        case 4: // 或者 case 4 ... 12
        default:
            // 多个峰
            if (is_FM()) {
                printf("检测到: FM 调频\n");
            } else {
                printf("信号复杂或噪声过大\n");
            }
            break;
    }
}
```
